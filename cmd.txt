    --input "/home/yaad/Dataset_final_run/June 2 Gate/gate_june2_yaad" \

    --input "/home/yaad/Dataset_final_run/June 2 Gate/gate_june2_yaad" \
    --output /home/yaad/Dataset_final_run/split_out \
    --train 0.85 --val 0.05 --test 0.10

# ─── tune_yolo.py — hyperparameter tuning (run before training) ──────────────
python tune_yolo.py \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --model yolo11n.pt \
    --epochs 30 --iterations 50 \
    --device 0 --batch 16 --imgsz 640

# ─── train_yolo26.py — train YOLO26n (nano, Jetson-ready) ────────────────────
python train_yolo26.py \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --model yolo26n.pt \
    --epochs 100 --patience 20 --imgsz 640 \
    --batch 16 --device 0 \
    --name yolo26n_custom

# ─── train_yolo26m.py — train YOLO26m (medium) ───────────────────────────────
python train_yolo26m.py \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --model yolo26m.pt \
    --epochs 100 --patience 20 --imgsz 640 \
    --batch 8 --device 0 \
    --name yolo26m_custom

# ─── train_yolo11.py — train YOLO11n ─────────────────────────────────────────
python train_yolo11.py \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --model yolo11n.pt \
    --epochs 100 --patience 20 --imgsz 640 \
    --batch 16 --device 0 \
    --name yolo11n_custom

# with hyperparameters from tune_yolo.py:
python train_yolo11.py \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --cfg runs/tune/tune_result/best_hyperparameters.yaml \
    --model yolo11n.pt \
    --epochs 100 --device 0

# ─── prune_yolo26.py — structured pruning of a trained YOLO26n ───────────────
python prune_yolo26.py \
    --weights runs/detect/runs/train/yolo26n_custom/weights/best.pt \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --pruning-ratio 0.30 --finetune-epochs 25 \
    --device 0 --name yolo26n_pruned

# ─── prune_to_nano.py — iteratively shrink YOLO26m → nano-equivalent ─────────
python prune_to_nano.py \
    --weights runs/detect/runs/train/yolo26m_custom/weights/best.pt \
    --data /home/yaad/Dataset_final_run/split_out/data.yaml \
    --rounds 2 --ratio-per-round 0.40 --finetune-epochs 20 \
    --device 0 --name yolo26m_to_nano

# ─── benchmark.py — compare base vs pruned model ─────────────────────────────
python benchmark.py \
    --base    runs/detect/runs/train/yolo26n_custom/weights/best.pt \
    --pruned  runs/detect/runs/prune/yolo26n_pruned/weights/best.pt \
    --data    /home/yaad/Dataset_final_run/split_out/data.yaml \
    --device 0
